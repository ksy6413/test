Option Explicit

Public Sub EmptyItemsInAlertAndDirectChildren()
    Dim ns As Outlook.NameSpace
    Dim alertFolder As Outlook.Folder

    Set ns = Application.GetNamespace("MAPI")
    Set alertFolder = ns.PickFolder 'Pick your "Alert" folder
    If alertFolder Is Nothing Then Exit Sub

    DeleteAllItemsInFolder alertFolder

    Dim child As Outlook.Folder
    For Each child In alertFolder.Folders
        DeleteAllItemsInFolder child
    Next child
End Sub

Private Sub DeleteAllItemsInFolder(ByVal f As Outlook.Folder)
    Dim i As Long
    For i = f.Items.Count To 1 Step -1
        f.Items(i).Delete   'moves to Deleted Items
    Next i
End Sub
