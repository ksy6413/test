    // ============================================================================
    // Focus recovery: Escape from workbench regions back to the active editor
    // ============================================================================

    // Primary Side Bar / Secondary Side Bar / Panel → editor.
    // Keep terminal excluded so shell tools such as vim, less, and fzf still receive Escape.
    {
        "key": "escape",
        "command": "workbench.action.focusActiveEditorGroup",
        "when": "sideBarFocus && !inputFocus && !suggestWidgetVisible && !findWidgetVisible"
    },
    {
        "key": "escape",
        "command": "workbench.action.focusActiveEditorGroup",
        "when": "auxiliaryBarFocus && !inputFocus"
    },
    {
        "key": "escape",
        "command": "workbench.action.focusActiveEditorGroup",
        "when": "panelFocus && !terminalFocus && !inputFocus"
    },

    // ============================================================================
    // Editor line operations
    // ============================================================================

    // Move lines with Shift+Alt+Up/Down; free Alt+Up/Down for other use.
    {
        "key": "shift+alt+down",
        "command": "editor.action.moveLinesDownAction",
        "when": "editorTextFocus && !editorReadonly"
    },
    {
        "key": "shift+alt+up",
        "command": "editor.action.moveLinesUpAction",
        "when": "editorTextFocus && !editorReadonly"
    },
    {
        "key": "shift+alt+down",
        "command": "-editor.action.copyLinesDownAction",
        "when": "editorTextFocus && !editorReadonly"
    },
    {
        "key": "shift+alt+up",
        "command": "-editor.action.copyLinesUpAction",
        "when": "editorTextFocus && !editorReadonly"
    },
    {

        "command": "-editor.action.moveLinesDownAction",

    },
    {

    },

    // Delete current line with Ctrl+Shift+D. Use editorTextFocus, not textInputFocus,
    // to avoid surprising behavior in non-editor text inputs.
    {
        "key": "ctrl+shift+d",
        "command": "editor.action.deleteLines",
        "when": "editorTextFocus && !editorReadonly"

    {
        "key": "ctrl+shift+k",

        "key": "ctrl+shift+d",
        "command": "-workbench.view.debug",
        "when": "viewContainer.workbench.view.debug.enabled"
    },
    {
        "key": "ctrl+shift+d",
        "command": "-extension.debugMsgBox",
        "when": "editorTextFocus && editorLangId == 'autoit'"
    },

    // Multi-cursor: use Alt+J for "Add Selection to Next Find Match".
    {
        "key": "alt+j",
        "command": "editor.action.addSelectionToNextFindMatch",

    {

        "when": "config.notebook.multiCursor.enabled && notebookCellEditorFocused && activeEditor == 'workbench.editor.notebook'"
